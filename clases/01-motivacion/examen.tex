\begin{questions}

\question[1] Seg\'un la clase, ?`cu\'ales son los tres pilares sobre los que se
sostiene la inteligencia artificial?
\begin{choices}
  \choice Algoritmos, hardware y presupuesto.
  \CorrectChoice Datos, infraestructura y personas.
  \choice Estad\'istica, programaci\'on y matem\'aticas.
  \choice Modelos, m\'etricas y servidores.
\end{choices}

\question[1] Hoy el principal cuello de botella para hacer IA, seg\'un la clase,
ya no es el algoritmo ni el c\'omputo, sino:
\begin{choices}
  \choice El lenguaje de programaci\'on elegido.
  \choice El n\'umero de par\'ametros del modelo.
  \CorrectChoice Los datos de calidad.
  \choice La velocidad de la red.
\end{choices}

\question[1] Si una c\'amara captura una imagen de 640x640 p\'ixeles en bruto,
?`a qu\'e nivel de la pir\'amide DIKW corresponde por s\'i sola?
\begin{choices}
  \CorrectChoice Dato.
  \choice Informaci\'on.
  \choice Conocimiento.
  \choice Sabidur\'ia.
\end{choices}

\question[1] La frase ``garbage in, garbage out'' aplicada a la IA quiere decir
que:
\begin{choices}
  \choice Los modelos siempre descartan los datos ruidosos autom\'aticamente.
  \choice Mientras m\'as datos haya, mejor sin importar su calidad.
  \choice El c\'omputo compensa cualquier deficiencia de los datos.
  \CorrectChoice Un modelo es tan bueno como los datos con los que aprende.
\end{choices}

\question[1] ?`Cu\'al es la diferencia esencial entre la programaci\'on cl\'asica
y el machine learning?
\begin{choices}
  \choice La programaci\'on cl\'asica usa datos y el ML no.
  \CorrectChoice En la cl\'asica el humano escribe las reglas; en ML el algoritmo
    las descubre a partir de ejemplos.
  \choice El ML solo sirve para datos tabulares peque\~nos.
  \choice La programaci\'on cl\'asica aprende de la experiencia y el ML no.
\end{choices}

\question[1] Se mencionan el ajedrez y el Go como problemas donde la fuerza bruta
no alcanza. Esto se debe principalmente a que:
\begin{choices}
  \choice Sus reglas cambian en cada partida.
  \choice No existen ejemplos de partidas previas.
  \CorrectChoice Su espacio de b\'usqueda es gigantesco (p.ej. Go ~ $10^{170}$
    posiciones).
  \choice Requieren hardware cu\'antico para resolverse.
\end{choices}

\question[1] ?`Cu\'al es la relaci\'on correcta entre Inteligencia Artificial
(IA), Machine Learning (ML) y Deep Learning (DL)?
\begin{choices}
  \choice Son tres campos independientes sin relaci\'on.
  \choice DL contiene a ML y ML contiene a IA.
  \CorrectChoice IA contiene a ML y ML contiene a DL.
  \choice IA y ML son lo mismo, y DL es algo distinto.
\end{choices}

\question[1] ?`Por qu\'e 2012 se describe como ``el a\~no en que los datos
cambiaron todo''?
\begin{choices}
  \CorrectChoice AlexNet, entrenada con el dataset masivo ImageNet y GPUs, redujo
    dr\'asticamente el error y deton\'o la era del deep learning.
  \choice Se public\'o el primer lenguaje de programaci\'on para IA.
  \choice Deep Blue venci\'o a Kasparov en ajedrez.
  \choice Se otorg\'o el primer premio Nobel a la IA.
\end{choices}

\question[1] Una diferencia clave entre el ML cl\'asico y el deep learning es
que en el deep learning:
\begin{choices}
  \choice El humano dise\~na manualmente los atributos de entrada.
  \choice Se requieren menos datos y menos c\'omputo.
  \CorrectChoice La red aprende los atributos por s\'i misma (extremo a extremo),
    pero exige muchos m\'as datos y c\'omputo.
  \choice Solo funciona con datos tabulares peque\~nos.
\end{choices}

\question[1] En la taxonom\'ia del machine learning, segmentar clientes sin
contar con etiquetas previas (clustering) es un ejemplo de aprendizaje:
\begin{choices}
  \choice Supervisado.
  \choice Por refuerzo.
  \choice Semi-supervisado.
  \CorrectChoice No supervisado.
\end{choices}

\question[1] El Premio Turing 2018 reconoci\'o a Bengio, Hinton y LeCun, y el
Nobel 2024 volvi\'o a premiar a Hinton (y a Hopfield). ?`Por qu\'e avance
principalmente?
\begin{choices}
  \choice Por inventar el primer lenguaje de programaci\'on.
  \CorrectChoice Por los avances que hicieron del aprendizaje con redes
    neuronales un motor de la computaci\'on y la ciencia.
  \choice Por dise\~nar la arquitectura de las GPUs.
  \choice Por crear el dataset ImageNet.
\end{choices}

\question[1] Preparar datos para que sean ``AI-ready'' agrega etapas nuevas
frente a la gesti\'on tradicional de datos. ?`Cu\'al de las siguientes es una de
esas etapas mencionadas en la clase?
\begin{choices}
  \choice Normalizaci\'on de tablas relacionales \'unicamente.
  \choice Respaldo en cintas magn\'eticas.
  \CorrectChoice Etiquetado, embeddings, chunking y mitigaci\'on de sesgos.
  \choice Eliminaci\'on de todos los datos no estructurados.
\end{choices}

\end{questions}
