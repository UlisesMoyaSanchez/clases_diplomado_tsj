\begin{questions}

\question[1] ?`Que etapas del ciclo de vida de los datos se estudian en esta
clase (Parte II)?
\begin{choices}
  \choice Creaci\'on, captura y almacenamiento.
  \CorrectChoice Acceso/seguridad, an\'alisis/visualizaci\'on y
    archivado/eliminaci\'on.
  \choice Solo la limpieza de datos.
  \choice Solo la captura por censo y encuesta.
\end{choices}

\question[1] La mejor defensa contra una \textbf{inyecci\'on SQL} es:
\begin{choices}
  \choice Guardar la base de datos en formato .xls.
  \choice Dar privilegios de administrador a todos los usuarios.
  \CorrectChoice Usar consultas parametrizadas y validar/sanitizar las entradas.
  \choice Concatenar la entrada del usuario directamente en la consulta.
\end{choices}

\question[1] Se habla de proteger los datos ``en reposo'' y ``en tr\'ansito''.
?`A que se refiere ``en tr\'ansito''?
\begin{choices}
  \CorrectChoice A los datos mientras viajan por la red entre sistemas.
  \choice A los datos guardados en disco sin moverse.
  \choice A los datos que ya fueron eliminados.
  \choice A los datos impresos en papel.
\end{choices}

\question[1] El principio de \textbf{m\'inimos privilegios} (RBAC) consiste en:
\begin{choices}
  \choice Cifrar todos los datos con la misma clave.
  \choice Hacer publicos los datos para mayor transparencia.
  \CorrectChoice Dar a cada usuario solo el acceso que necesita para su tarea.
  \choice Eliminar las contrase\~nas para agilizar el acceso.
\end{choices}

\question[1] Usar IA para crear audio o video falso que suplanta a una persona
real se conoce como:
\begin{choices}
  \choice Data poisoning.
  \CorrectChoice Deepfake.
  \choice Inyecci\'on SQL.
  \choice Hardening.
\end{choices}

\question[1] En el an\'alisis exploratorio, la regla \textbf{1.5 x IQR} sirve
para:
\begin{choices}
  \choice Rellenar los valores faltantes.
  \choice Cifrar los datos sensibles.
  \CorrectChoice Detectar valores at\'ipicos (outliers).
  \choice Reducir la dimensionalidad de los datos.
\end{choices}

\question[1] El \textbf{cuarteto de Anscombe} demuestra que:
\begin{choices}
  \CorrectChoice Conjuntos con las mismas estad\'isticas pueden tener formas muy
    distintas, por eso hay que visualizar.
  \choice Los datos siempre siguen una linea recta.
  \choice La media es suficiente para describir cualquier conjunto.
  \choice Visualizar los datos es innecesario si se conoce la correlaci\'on.
\end{choices}

\question[1] ?`Cu\'al de estas tecnicas se usa para \textbf{reducir la
dimensionalidad} de un conjunto con muchas variables?
\begin{choices}
  \choice IQR.
  \choice TLS.
  \CorrectChoice PCA (o t-SNE, UMAP).
  \choice RBAC.
\end{choices}

\question[1] Para anonimizar datos personales antes de compartirlos, una buena
practica es:
\begin{choices}
  \choice Ordenarlos alfabeticamente por nombre.
  \CorrectChoice Reemplazar la identidad por un seudonimo (p.ej. un hash
    SHA-256) y enmascarar campos sensibles.
  \choice Convertir el archivo a JSON.
  \choice Subirlos a una hoja de calculo compartida.
\end{choices}

\question[1] Un usuario pide que borren sus datos personales porque ya no quiere
ser cliente. ?`Que derecho ARCO esta ejerciendo?
\begin{choices}
  \choice Acceso.
  \choice Rectificaci\'on.
  \CorrectChoice Cancelaci\'on.
  \choice Oposici\'on.
\end{choices}

\question[1] Una buena practica al \textbf{archivar} datos inactivos de larga
estancia es:
\begin{choices}
  \choice Dejarlos en la base de produccion sin cambios.
  \choice Borrar todos los metadatos para ahorrar espacio.
  \CorrectChoice Moverlos a almacenamiento frio, comprimirlos/cifrarlos y
    documentar sus metadatos (fecha, responsable, razon).
  \choice Guardar una sola copia sin ningun registro de la operacion.
\end{choices}

\end{questions}
