\begin{frame}{Datos cualitativos (categoricos)}
  \begin{itemize}
    \item \textbf{Nominal}: categorias sin orden. Solo se pueden comparar por
          igualdad. Ej.: color de ojos, pais, tipo de sangre.
    \item \textbf{Ordinal}: categorias con un orden, pero sin distancia definida.
          Ej.: nivel educativo, satisfaccion (bajo/medio/alto), severidad.
    \item \textbf{Binario / booleano}: caso especial con dos categorias
          (si/no, verdadero/falso, 0/1).
  \end{itemize}
  \begin{block}{Cuidado}
    Codificar ``bajo=1, medio=2, alto=3'' NO los vuelve numeros: la diferencia
    entre niveles no es comparable como en una escala continua.
  \end{block}
\end{frame}

\begin{frame}{Datos cuantitativos (numericos)}
  \begin{itemize}
    \item \textbf{Discreto}: valores contables, normalmente enteros.
          Ej.: numero de hijos, visitas a una pagina, defectos por lote.
    \item \textbf{Continuo}: cualquier valor en un intervalo (se mide).
          Ej.: estatura, temperatura, tiempo de respuesta.
  \end{itemize}
  \vspace{0.3em}
  Escalas de medicion clasicas (Stevens):
  \begin{itemize}
    \item \textbf{Intervalo}: sin cero absoluto (grados Celsius, anio).
    \item \textbf{Razon}: con cero absoluto (peso, ingreso, distancia).
  \end{itemize}
\end{frame}

\begin{frame}{Tipos especiales: fechas, tiempo y texto}
  \begin{itemize}
    \item \textbf{Fecha / hora}: parecen texto, pero tienen orden y aritmetica
          (restar fechas da una duraci\'on). Usar siempre formato ISO 8601:
          \texttt{2026-06-29}, no \texttt{29/06/26}.
    \item \textbf{Series de tiempo}: secuencia de valores indexada por instante.
    \item \textbf{Geoespacial}: coordenadas (lat, lon), poligonos.
    \item \textbf{Texto libre}: cadena sin estructura fija (un comentario, un
          articulo) -- es el puente hacia los datos no estructurados.
  \end{itemize}
\end{frame}

  \end{columns}

\section{Los datos y su representaci\'on}

\begin{frame}{Al final, todo se vuelve numeros}
  Para que una computadora (y un modelo) procese un dato, hay que
  \textbf{representarlo} como numeros:
  \begin{itemize}
    \item \textbf{Tabla}: matriz filas x columnas.
    \item \textbf{Imagen}: matriz de pixeles (alto x ancho x canales RGB).
    \item \textbf{Audio}: vector de amplitudes muestreadas en el tiempo.
    \item \textbf{Texto}: secuencia de tokens -> vectores (embeddings).
    \item \textbf{Grafo}: nodos + aristas (matriz de adyacencia).
  \end{itemize}
  La representaci\'on elegida condiciona qu\'e modelo se puede usar.
\end{frame}

\begin{frame}[fragile]{Una imagen es una matriz de pixeles}
  \begin{columns}[c]
    \column{0.5\textwidth}
    \begin{center}
    \begin{tikzpicture}[scale=0.42]
      \foreach \x in {0,...,7}
        \foreach \y in {0,...,7} {
          \pgfmathsetmacro{\g}{(\x+\y)*6}
          \fill[black!\g] (\x,-\y) rectangle ++(1,-1);
          \draw[gray!40] (\x,-\y) rectangle ++(1,-1);
        }
    \end{tikzpicture}
    \end{center}
    \column{0.5\textwidth}
      \begin{itemize}
        \item Cada celda es un \textbf{pixel}: un numero 0--255 por canal.
        \item Gris: una matriz $H\times W$. Color: tres matrices (R, G, B).
        \item Una foto de 640x640 a color son
              $640\cdot640\cdot3 \approx 1.2$ millones de numeros.
      \end{itemize}
      Por eso las im\'agenes son ``pesadas'' y necesitan redes especializadas
      (CNN).
  \end{columns}
\end{frame}

\section{Gram\'atica de los datos: tidy data}

\begin{frame}{?`Qu\'e es la gramatica de los datos?}
  As\'i como una lengua tiene reglas para formar frases claras, los datos
  tabulares tienen una \textbf{gramatica} para ser ordenados y reutilizables.
  El estandar es \emph{tidy data} (Hadley Wickham):
  \begin{enumerate}
    \item Cada \textbf{variable} es una \textbf{columna}.
    \item Cada \textbf{observacion} es una \textbf{fila}.
    \item Cada \textbf{tipo de unidad observacional} es una \textbf{tabla}.
  \end{enumerate}
  \begin{block}{Por que importa}
    Datos ordenados = una sola forma correcta; datos desordenados = mil formas
    distintas. Lo ordenado se filtra, agrupa y grafica sin pelear.
  \end{block}
\end{frame}

\begin{frame}[fragile]{Formato ancho (wide) vs ordenado (tidy)}
  \begin{columns}[t]
    \column{0.5\textwidth}
      \textbf{Wide} (a\~nos como columnas)
\begin{lstlisting}[language=,basicstyle=\ttfamily\scriptsize]
ciudad | 2020 | 2021 | 2022
GDL    | 28.1 | 28.7 | 29.0
CDMX   | 17.5 | 17.9 | 18.2
\end{lstlisting}
      Comodo para leer, malo para procesar:
      el a\~no es un \textbf{valor}, no una variable.
    \column{0.5\textwidth}
      \textbf{Tidy} (una observacion por fila)
\begin{lstlisting}[language=,basicstyle=\ttfamily\scriptsize]
ciudad | anio | temp
GDL    | 2020 | 28.1
GDL    | 2021 | 28.7
GDL    | 2022 | 29.0
CDMX   | 2020 | 17.5
...
\end{lstlisting}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{De wide a tidy con pandas}
\begin{lstlisting}[language=Python]
import pandas as pd

# wide: una columna por anio
df = pd.read_csv("temperaturas_wide.csv")

# melt: "alarga" la tabla a formato tidy
tidy = df.melt(id_vars="ciudad",
               var_name="anio",
               value_name="temp")

tidy["anio"] = tidy["anio"].astype(int)
\end{lstlisting}
  \vspace{0.2em}
  \texttt{melt} ordena (wide -> long); \texttt{pivot} hace lo inverso
  (long -> wide) cuando se necesita una tabla de reporte.
\end{frame}

\section{Malas pr\'acticas en hojas de c\'alculo}

\begin{frame}{Reglas de oro para una tabla sana}
  \begin{columns}[t]
    \column{0.5\textwidth}
      \textbf{Si}
      \begin{itemize}
        \item Una variable por columna.
        \item Una observaci\'on por fila.
        \item Encabezados en la 1a fila, cortos y sin espacios.
        \item Un solo dato por celda.
        \item Codigos de faltante explicitos (NA).
      \end{itemize}
    \column{0.5\textwidth}
      \textbf{No}
      \begin{itemize}
        \item Celdas combinadas (merge).
        \item Colores como si fueran datos.
        \item Varias tablas en una hoja.
        \item Totales mezclados con datos.
        \item Unidades dentro de la celda (``34 kg'').
      \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Malas practicas tipicas (y su arreglo)}
  \begin{itemize}
    \item \textbf{Columnas que inician con numero o espacio}: \texttt{2020},
          \texttt{1er trimestre}. Muchas herramientas no las aceptan como nombre.
          Arreglo: \texttt{anio\_2020}, \texttt{trim\_1}.
    \item \textbf{Los datos crecen en filas, no en columnas}: agregar un a\~no
          nuevo NO debe ser una columna nueva, sino filas nuevas (formato tidy).
    \item \textbf{Fechas como texto} (\texttt{29/06/26}): ambiguas y no ordenables.
          Arreglo: ISO \texttt{2026-06-29}.
    \item \textbf{Nombres que el programa ``corrige'' solo}: ver el caso de los
          genes en la siguiente secci\'on.
  \end{itemize}
\end{frame}

\section{Cuando Excel sale caro}

\begin{frame}{Catastrofes reales por hojas de calculo (I)}
  \begin{block}{Reinhart \& Rogoff (2010)}
    Un paper muy citado sobre deuda y crecimiento ten\'ia un rango de celdas
    mal seleccionado en Excel: omiti\'o 5 paises. El error reforz\'o pol\'iticas
    de \textbf{austeridad} antes de ser detectado por un estudiante en 2013.
  \end{block}
  \begin{block}{Public Health England (2020)}
    Cerca de \textbf{16,000 casos de COVID-19} no se reportaron porque se uso
    el formato viejo \texttt{.xls}, limitado a 65,536 filas: los contagios
    extra simplemente se truncaron y no se rastrearon contactos.
  \end{block}
\end{frame}

\begin{frame}{Catastrofes reales por hojas de calculo (II)}
  \begin{itemize}
    \item \textbf{Genes convertidos en fechas}: Excel transformaba \texttt{SEPT2}
          en ``2-Sep'' y \texttt{MARCH1} en fecha. Un estudio (2016) hall\'o
          errores en \textbf{1 de cada 5} papers genomicos; en 2020 el comite
          HGNC \textbf{renombro genes} para esquivar a Excel.
    \item \textbf{JPMorgan ``London Whale'' (2012)}: un modelo de riesgo en Excel
          con copiar/pegar manual entre celdas contribuyo a perdidas de
          \textbf{6 mil millones} de dolares.
    \item \textbf{TransAlta (2003)}: un error de copiar/pegar en una hoja le costo
          \textbf{24 millones} de dolares en contratos mal asignados.
  \end{itemize}
  \vspace{0.3em}
  Moraleja: la hoja de calculo es genial para mirar, peligrosa como base de datos.
\end{frame}

\section{Resumen}

\begin{frame}{Takeaways}
  \begin{itemize}
    \item Los datos se clasifican por \textbf{naturaleza} (cualitativos vs
          cuantitativos; nominal/ordinal, discreto/continuo) y por
          \textbf{estructura} (estructurados, semi, no estructurados).
    \item Todo dato -- imagen, audio, texto, grafo -- termina representado como
          \textbf{numeros}; la representaci\'on define el modelo aplicable.
    \item La \textbf{gramatica tidy}: una variable por columna, una observaci\'on
          por fila, una tabla por unidad.
    \item Las hojas de calculo invitan a malas pr\'acticas; varios desastres
          (Reinhart-Rogoff, PHE, genes, JPMorgan) nacieron de ah\'i.
    \item Datos ordenados hoy = menos sufrimiento (y menos catastrofes) ma\~nana.
  \end{itemize}
\end{frame}
